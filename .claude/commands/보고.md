# 📊 보고 - 통합 보고서 생성 시스템 v31.0

## 📋 Claude 실행 프로토콜

```python
# 보고 단계 자동 실행 로직
def execute_reporting(arguments):
    """통합 보고서 생성 시스템 실행 (폴백 시스템 포함)"""
    
    # 1. 의존성 자동 확인 및 설치
    auto_report_path = ensure_auto_report_generator()
    
    if auto_report_path:
        # Auto Report Generator 사용
        import subprocess
        result = subprocess.run([
            'python', auto_report_path,
            '--title', arguments or '프로젝트 진행 보고서'
        ], capture_output=True, text=True)
        
        if result.returncode == 0:
            print("📊 자동 보고서 생성 완료!")
            print(f"📄 위치: {result.stdout.strip()}")
            return {'status': 'success', 'report_path': result.stdout.strip()}
        else:
            print("⚠️ 자동 보고서 생성 실패, 기본 보고서로 전환")
    
    # 2. 폴백: 기본 보고서 생성
    basic_report = generate_basic_report(arguments)
    print("📊 기본 보고서 생성 완료!")
    return {'status': 'basic', 'report': basic_report}

def ensure_auto_report_generator():
    """auto_report_generator.py 존재 확인 및 자동 다운로드"""
    import os
    from pathlib import Path
    
    # 1차: 로컬에서 찾기
    script_path = Path('scripts/auto_report_generator.py')
    if script_path.exists() and script_path.stat().st_size > 1000:
        return str(script_path)
    
    # 2차: 자동 다운로드 시도
    print("📥 보고서 생성 엔진 다운로드 중...")
    os.makedirs('scripts', exist_ok=True)
    
    try:
        import urllib.request
        urllib.request.urlretrieve(
            "https://raw.githubusercontent.com/kyuwon-shim-ARL/claude-dev-kit/main/scripts/auto_report_generator.py",
            script_path
        )
        if script_path.exists() and script_path.stat().st_size > 1000:
            print("✅ 보고서 생성 엔진 설치 완료")
            return str(script_path)
    except Exception as e:
        print(f"⚠️ 다운로드 실패: {e}")
    
    # 3차: 폴백 - None 반환
    return None

def generate_basic_report(arguments):
    """기본 보고서 생성 (자동화 없이)"""
    import datetime
    
    timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M")
    title = arguments or "프로젝트 기본 보고서"
    
    return f"""# {title}

**생성일시**: {timestamp}
**생성방식**: 기본 보고서 (자동 분석 미사용)

## 📋 요약
이 보고서는 자동 분석 도구 없이 생성된 기본 보고서입니다.
완전한 자동 보고서를 위해서는 `scripts/auto_report_generator.py` 파일이 필요합니다.

## 🔧 완전한 자동 보고서 설정 방법

### 1. claude-dev-kit 재설치
```bash
./init.sh --upgrade  # 기존 프로젝트에서 실행
```

### 2. 수동 스크립트 다운로드
```bash
mkdir -p scripts
curl -sSL "https://raw.githubusercontent.com/kyuwon-shim-ARL/claude-dev-kit/main/scripts/auto_report_generator.py" \\
     -o scripts/auto_report_generator.py
chmod +x scripts/auto_report_generator.py
```

### 3. 다시 실행
```bash
/보고  # 이제 완전한 자동 보고서 생성됩니다
```

---
*완전한 기능을 위해 claude-dev-kit을 최신 버전으로 업그레이드하세요.*
"""

# 즉시 실행
if __name__ == "__main__":
    execute_reporting(ARGUMENTS)
```
